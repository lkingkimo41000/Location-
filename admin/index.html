function submitCompleteRes() { const r = db.reservations.find(r => r.id === state.modal.resId); if (!r) return; const km = Number(document.getElementById('complete-km').value); if (!km || km <= 0) { return showToast('أدخل قراءة الكيلومترات (أكبر من 0)', 'error'); } const car = db.cars.find(c => c.id === r.carId); if (car) { car.status = 'available'; updateCarKm(car.id, km); } // تغيير حالة الحجز إلى "منتهي" r.status = 'completed'; saveData(); state.modal = null; render(); showToast('✅ تم إتمام الإرجاع - الحجز منتهي', 'success'); }
